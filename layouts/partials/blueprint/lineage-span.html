"page" $b "href" $href) }}{{ end }} {{ end }} {{ end }} {{ $out := false }} {{ with $entries }} {{ $entries = sort . "year" }} {{ $seen := slice }} {{ $marked := slice }} {{ range $entries }} {{ $marked = $marked | append (merge . (dict "again" (in $seen .page.RelPermalink))) }} {{ $seen = $seen | append .page.RelPermalink }} {{ end }} {{ $entries = $marked }} {{ $first := (index $entries 0).year }} {{ $last := (index $entries (sub (len $entries) 1)).year }} {{ $start := mul (div $first 10) 10 }} {{ $end := add (mul (div $last 10) 10) 10 }} {{/* A year in the first or last few of its decade would sit at the very end of the track, its label hanging off it: add a decade there. */}} {{ if lt (sub $first $start) 3 }}{{ $start = sub $start 10 }}{{ end }} {{ if lt (sub $end $last) 3 }}{{ $end = add $end 10 }}{{ end }} {{ if eq (sub $end $start) 10 }}{{ $start = sub $start 10 }}{{ $end = add $end 10 }} {{ else if eq (sub $end $start) 20 }}{{ $end = add $end 10 }}{{ end }} {{ $out = dict "entries" $entries "start" $start "end" $end }} {{ end }} {{ return $out }}
