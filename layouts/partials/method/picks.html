{{/* A build page's picks, as a dict from part name (lower case) to the pick: `picks` is a list of "Part: pick" strings, split at the first colon (a list, not a map, because the content tools read flat frontmatter). A string with no colon is skipped. Takes a page. */}} {{ $out := dict }} {{ range .Params.picks | default slice }} {{ $bits := split (string .) ":" }} {{ $name := index $bits 0 | strings.TrimSpace }} {{ if and (gt (len $bits) 1) $name }} {{ $out = merge $out (dict (lower $name) (delimit (after 1 $bits) ":" | strings.TrimSpace)) }} {{ end }} {{ end }} {{ return $out }}
