{{/* Where a page sits in one of its blueprints (blueprint/info.html). Takes a dict, page and term (the blueprint's term page); returns a dict: info the blueprint's info kind "build" (a method with `build_name`), "part" (a method with `part`), "instance" (any other page with `build_name`, i.e. a recipe made from a build), or "member" (anything else) name the part or build name, "" for a member build for an instance, its build page, or false if none is written next the next build (for a build) or the next written part (for a part), or false n for a part, its number in the blueprint's `parts` order, and for the next part, nextN */}} {{ $page := .page }} {{ $info := partialCached "blueprint/info.html" .term .term.Path }} {{ $isMethod := partial "method/is.html" $page }} {{ $kind := "member" }} {{ $name := "" }} {{ $build := false }} {{ $next := false }} {{ $n := 0 }} {{ $nextN := 0 }} {{ if and $isMethod $page.Params.build_name }} {{ $kind = "build" }} {{ $name = string $page.Params.build_name }} {{ $seen := false }} {{ range $info.builds }} {{ if and $seen (not $next) }}{{ $next = . }}{{ end }} {{ if eq .RelPermalink $page.RelPermalink }}{{ $seen = true }}{{ end }} {{ end }} {{ else if and $isMethod $page.Params.part }} {{ $kind = "part" }} {{ $name = string $page.Params.part }} {{ range $info.parts }} {{ if and $n (not $next) .methods }}{{ $next = index .methods 0 }}{{ $nextN = .n }}{{ end }} {{ if eq (lower .name) (lower $name) }}{{ $n = .n }}{{ $name = .name }}{{ end }} {{ end }} {{ else if $page.Params.build_name }} {{ $kind = "instance" }} {{ $name = string $page.Params.build_name }} {{ range $info.groups }}{{ if eq (lower .name) (lower $name) }}{{ $build = .page }}{{ end }}{{ end }} {{ end }} {{ return dict "info" $info "kind" $kind "name" $name "build" $build "next" $next "n" $n "nextN" $nextN }}
