{{/* Whether a page is a method: a regular page in the methods section (params.fugu.methodSection, e.g. content/methods/), or in the essay section's params.fugu.methods.folder (content/essays/methods/ by default), at any depth. The folder decides, not the page's tags. Takes a page; returns a bool, false when neither is set up. */}} {{ $is := false }} {{ if and .IsPage .File }} {{ $methods := partial "section/name.html" "method" }} {{ $essays := partial "section/name.html" "essay" }} {{ $folder := site.Params.fugu.methods.folder | default "" }} {{ if and $methods (eq .Section $methods) }} {{ $is = true }} {{ else if and $essays $folder (eq .Section $essays) }} {{ $is = hasPrefix (replace .File.Dir "\\" "/") (printf "%s/%s/" $essays $folder) }} {{ end }} {{ end }} {{ return $is }}
