{{/* Part navigation at the foot of a blueprint's part or build, in place of the date-ordered previous/next (article-pagination.html): back to the blueprint on the left, and on the right the next part in weight order (for a part) or the next build in era order (for a build), from blueprint/info.html. Returns the HTML, or "" when the page isn't a part or a build, so the caller can fall back. Takes the page. */}} {{ $out := "" }} {{ $page := . }} {{ with partial "blueprint/of.html" . }} {{ if in (slice "part" "build") .role }} {{ $bp := .blueprint }} {{ $isBuild := eq .role "build" }} {{ $info := partialCached "blueprint/info.html" $bp $bp.Path }} {{ $next := false }} {{ $nextN := 0 }} {{ $seen := false }} {{ if $isBuild }} {{ range $info.builds }} {{ if and $seen (not $next) }}{{ $next = . }}{{ end }} {{ if eq . $page }}{{ $seen = true }}{{ end }} {{ end }} {{ else }} {{ range $info.parts }} {{ if and $seen (not $next) }}{{ $next = .page }}{{ $nextN = .n }}{{ end }} {{ if eq .page $page }}{{ $seen = true }}{{ end }} {{ end }} {{ end }} {{ $up := printf "%s blueprint" $bp.LinkTitle | htmlEscape }} {{ $nextHTML := "" }} {{ with $next }} {{ $label := cond $isBuild "Next build" (printf "Part %d" $nextN) }} {{ $nextHTML = printf `%s →%s` .RelPermalink $label (.Title | htmlEscape) }} {{ end }} {{ $out = printf `
← %s%s%s
` $up $bp.RelPermalink $up (cond $isBuild "All builds" "All parts") $nextHTML }} {{ end }} {{ end }} {{ return (safeHTML $out) }}
