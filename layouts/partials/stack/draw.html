{{/* Draws a ```stack block as an exploded view: each layer from assets/icons/layers/.svg, top to bottom, pulled apart, the upper layers drawn over the lower. Shared by render-codeblock-stack.html (the page) and stack-strip.html (cards and the build page's diagram). One layer per line, `key | label | role`, plus optional lines: caption: Drive-in deluxe repeat: 3-5 x3 (layers 3 to 5, three times: a "× 3" bracket) The layer files share a 210-wide viewBox centred on x=105, each with its own height; data-edge is how far the drawing reaches right of centre, so a leader line starts just past it. They're inlined, so currentColor and the theme tokens (--accent-soft, --surface) follow dark mode. Labels, leaders and the bracket carry plain defaults in currentColor; a site's CSS restyles them by class (.stack-label, .stack-role, .stack-lead, .stack-dot, .stack-bracket, .stack-times, .stack-cap). Takes a dict: inner (the block's text), labels (draw leaders, labels and roles on the right), compact (for cards: tighter gaps, tighter still for tall stacks, 6 for up to six layers and 0.75 less for each layer after, down to 2.5; a `repeat:` draws its layers once, so it doesn't count), caption (show the caption), pos (the block's .Position; when set, an unknown key or a bad repeat line warns, as the formula block does). */}} {{- $cx := 105.0 }} {{- $rx := 70.0 }} {{- $compact := .compact | default false }} {{- $gap := 16.0 }} {{- $labels := .labels | default false }} {{- $pos := .pos }} {{- $caption := "" }} {{- $repeats := slice }} {{- $layers := slice }} {{- range split .inner "\n" }} {{- $line := trim . " \t\r" }} {{- if not $line }}{{ continue }}{{ end }} {{- if hasPrefix $line "caption:" }} {{- $caption = trim (strings.TrimPrefix "caption:" $line) " " }} {{- continue }} {{- end }} {{- if hasPrefix $line "repeat:" }} {{- $spec := trim (strings.TrimPrefix "repeat:" $line) " " }} {{- with findRESubmatch `^(\d+)(?:\s*-\s*(\d+))?\s*[x×]\s*(.+)$` $spec 1 }} {{- $m := index . 0 }} {{- $from := int (index $m 1) }} {{- $to := cond (ne (index $m 2) "") (int (index $m 2)) $from }} {{- $repeats = $repeats | append (dict "from" $from "to" $to "n" (index $m 3)) }} {{- else }} {{- with $pos }}{{ warnf "stack: can't read %q (try `repeat: 3-5 x3`) at %s" $line . }}{{ end }} {{- end }} {{- continue }} {{- end }} {{- $f := slice }} {{- range split $line "|" }}{{ $f = $f | append (trim . " ") }}{{ end }} {{- $key := index $f 0 }} {{- $layer := dict "key" $key "label" "" "role" "" "svg" "" "h" 20.0 "edge" 40.0 }} {{- if ge (len $f) 2 }}{{ $layer = merge $layer (dict "label" (index $f 1)) }}{{ end }} {{- if ge (len $f) 3 }}{{ $layer = merge $layer (dict "role" (index $f 2)) }}{{ end }} {{- with and (findRE `^[a-z0-9-]+$` $key) (resources.Get (printf "icons/layers/%s.svg" $key)) }} {{- $c := .Content }} {{- $h := index (index (findRESubmatch `viewBox="0 0 [0-9.]+ ([0-9.]+)"` $c 1) 0) 1 }} {{- $edge := index (index (findRESubmatch `data-edge="([0-9.]+)"` $c 1) 0) 1 }} {{- $layer = merge $layer (dict "svg" (replaceRE `(?s)^.*?]*>\s*|\s*\s*$` "" $c) "h" (float $h) "edge" (float $edge)) }} {{- else }} {{- with $pos }}{{ warnf "stack: no layer drawing %q (assets/icons/layers/) at %s" $key . }}{{ end }} {{- $layer = merge $layer (dict "svg" (printf `` $cx)) }} {{- end }} {{- $layers = $layers | append $layer }} {{- end }} {{- with $layers }} {{- if $compact }}{{ $gap = math.Max 2.5 (sub 6.0 (mul 0.75 (math.Max 0 (sub (len .) 6)))) }}{{ end }} {{- /* Place each layer below the last; draw bottom first, so upper layers sit over lower ones. */}} {{- $y := 8.0 }} {{- $placed := slice }} {{- $body := "" }} {{- range . }} {{- $placed = $placed | append (merge . (dict "y" $y "mid" (add $y (div .h 2)))) }} {{- $body = printf `%s%s` $y .svg $body }} {{- $y = add $y (add .h $gap) }} {{- end }} {{- $height := add (sub $y $gap) 8 }} {{- $labelX := add $cx (add (mul $rx 1.35) 16) }} {{- $bracketX := sub $cx (mul $rx 1.55) }} {{- $minX := cond (gt (len $repeats) 0) (sub $bracketX 34) 0.0 }} {{- $width := cond $labels (add $labelX 160) (mul $cx 2) }} {{- $names := slice }} {{- range $placed }}{{ $names = $names | append (.label | default .key) }}{{ end }} {{- end }}
