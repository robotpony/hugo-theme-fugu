{{/* One plain-text description per page, shared by the meta description (head.html) and the Open Graph / Twitter tags (social.html). Order: frontmatter `description`, then the page's intro (the prose before its first heading), then a fallback by page kind. Hugo's .Summary isn't used for pages: it cuts at summaryLength words and runs straight on into "Ingredients 2 tablespoons…" on recipes with a short intro. The intro drops blockquotes (family backstory, hidden by default), fenced code blocks (a formula or stack diagram would render, and warn, a second time), and shortcodes, and reduces [[wiki links]] to their text, since wiki links are only converted when single.html renders the body. */}} {{ $d := .Description }} {{ if and (not $d) .IsPage }} {{ $intro := replaceRE `(?ms)^#{1,6}\s.*` "" .RawContent }} {{ $intro = replaceRE `(?m)^>.*$` "" $intro }} {{ $intro = replaceRE "(?ms)^```.*?^```[ \t]*$" "" $intro }} {{ $intro = replaceRE `(?s)\{\{[<%].*?[%>]\}\}` "" $intro }} {{ $intro = replaceRE `\[\[(?:[^\]|]*\|)?([^\]]+)\]\]` "$1" $intro }} {{ $intro = $intro | markdownify | plainify | htmlUnescape }} {{ $intro = replaceRE `\s+` " " $intro | strings.TrimSpace }} {{/* Too short to describe anything: a food log month's "_Undated._" */}} {{ if ge (len $intro) 20 }} {{ $d = $intro }} {{ else if eq (partialCached "section/role.html" .Section .Section) "recipe" }} {{ $d = i18n "fugu.description.recipe" (dict "Title" .Title "Site" site.Title) }} {{ else if eq (partialCached "section/role.html" .Section .Section) "log" }} {{/* A month of the log is a run of ### entries; list them. */}} {{ $entries := slice }} {{ range findRESubmatch `(?m)^#{2,3}\s+(.+)$` .RawContent }} {{ $entries = $entries | append (index . 1 | plainify | strings.TrimSpace) }} {{ end }} {{ $d = i18n "fugu.description.log" (dict "Title" .Title "Entries" (delimit $entries ", ")) }} {{ end }} {{ end }} {{ if and (not $d) (eq .Kind "term") }} {{ if eq .Data.Plural "cuisine" }} {{ $d = i18n "fugu.description.cuisine" (dict "Cuisine" .Title "Site" site.Title) }} {{ else }} {{ $d = i18n "fugu.description.term" (dict "Term" .Title "Site" site.Title) }} {{ end }} {{ end }} {{ $d = $d | default site.Params.description }} {{ return (truncate 160 $d) }}
