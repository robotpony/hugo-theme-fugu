{{/* Formula strip: a page's first ```formula block drawn as one small row of icons and operators, for its card (recipe-card.html). Takes a Page. Cards don't render the body, so this finds the block in .RawContent and reads it with hugo-warped's parse (partials/fence/parse.html), keeping only the operator and icon keys, plus the quantity for a `:` ratio, where the numbers are the point. Labels, swaps, the `=` result, caption and bar never show here. Outputs nothing when the page has no formula block. It never warns: the page's own hook already has. Each slot carries the rest in a native tooltip (title): label, quantity, and swaps on a second line. The card's stretched title link would cover the strip and swallow the hover, so the strip's contents sit in their own link to the page, lifted above it (custom.css), kept out of the tab order and hidden from screen readers, which get the role="img" label. */}} {{- with findRESubmatch "(?s)```formula[^\\n]*\\n(.*?)\\n```" .RawContent 1 }} {{- $p := partial "fence/parse.html" (dict "type" "formula" "inner" (index (index . 0) 1)) }} {{- $slots := slice }} {{- $kind := "" }} {{- range $p.lines }} {{- if or .keyword (eq .marker "=") }}{{ continue }}{{ end }} {{- $op := .marker | default "" }} {{- if and $op (not $kind) }}{{ $kind = $op }}{{ end }} {{- $label := index .row 1 }} {{- $qty := "" }} {{- $swaps := "" }} {{- if ge (len .row) 3 }}{{ $qty = index .row 2 }}{{ end }} {{- if ge (len .row) 4 }}{{ $swaps = index .row 3 }}{{ end }} {{- $tip := $label }} {{- with $qty }}{{ $tip = cond (ne $tip "") (printf "%s · %s" $tip .) . }}{{ end }} {{- with $swaps }}{{ $tip = printf "%s\n%s" $tip (cond (hasPrefix . "or ") . (printf "or %s" .)) }}{{ end }} {{- $slots = $slots | append (dict "op" $op "icons" .icons "label" $label "qty" $qty "tip" $tip) }} {{- end }} {{- $ratio := eq $kind ":" }} {{- $words := dict "+" "plus" ":" "to" "→" "then" }} {{- $aria := slice }} {{- range $slots }} {{- $part := cond $ratio (trim (printf "%s %s" .label .qty) " ") .label }} {{- with .op }}{{ $part = printf "%s %s" (index $words .) $part }}{{ end }} {{- $aria = $aria | append $part }} {{- end }} {{- with $slots }}
