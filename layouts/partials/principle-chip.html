{{/* Principle chip: one per principle tag the given page carries (a tag whose content/tags//_index.md says `principle: true`, see principle/term.html), nothing otherwise. Takes a dict: page (required), compact (bool: uses the tag's `short` label, e.g. "WTF", for the recipe card's tight top row; full pages use the tag's title, e.g. "Win the fridge"). The icon is the tag's own (`icon`, see principle/term.html), else the cycle arrows. A tag with `card_label: false` is its icon alone on a card (.is-icon), its title in `title` and `aria-label`. On a page (not compact) it also carries a popover (the shared .pop-host/.pop component, see assets/js/popover.js) with the term's definition from the tag's essay, so hovering or focusing the chip explains it without a trip to the tag page. It opens downward, since the chip sits at the start of the page's tags row. On a card there's no popover, just the link (concept/pop.html). The chip and popover are the shared concept ones (concept/pop.html, concept/icon.html). */}} {{ $page := .page }} {{ $compact := .compact }} {{ range $page.Params.tags }} {{ with partialCached "principle/term.html" . . }} {{ $label := cond $compact .short .page.Title }} {{ $essay := .essay }} {{ $def := "" }} {{ with $essay }} {{ $def = .Params.summary | default .Description }} {{ if not $def }} {{ $head := index (split .RawContent "\n## ") 0 | strings.TrimSpace }} {{ if and $head (not (hasPrefix $head "#")) }} {{ $def = index (split $head "\n\n") 0 }} {{ end }} {{ end }} {{ $def = $def | markdownify | plainify | htmlUnescape | truncate 200 }} {{ end }} {{ $chip := printf `%s%s` .page.RelPermalink .icon $label }} {{ if and $compact (not .cardLabel) }} {{ $name := .page.Title | htmlEscape }} {{ $chip = printf `%s` .page.RelPermalink $name $name .icon }} {{ end }} {{ if $def }} {{ partial "concept/pop.html" (dict "kind" "principle" "card" $compact "trigger" $chip "head" (.page.Title | htmlEscape) "text" $def "link" $essay "more" "Read the full essay") }} {{ else }} {{ $chip | safeHTML }} {{ end }} {{ end }} {{ end }}
