{{/* Formula diagrams: a ```formula fenced block drawn as one row of icons joined by +, : or →, with the label and quantity under each icon. One slot per line, `icon | label | quantity | swaps`, with the operator at the start of every line after the first: can | Beans | 1 can | black, pinto, chickpea + bowl | Starch | 1 cup | rice, mash, quinoa + crumbs egg | Binder | ½ cup + 1 egg | or gluten + oats = patty | Patties | 6 caption: Per can of beans bar: yes One operator per block (+, : or →; -> works too), plus an optional closing `=` result. Icons are assets/icons/formula/.svg, inlined so they pick up currentColor and the theme tokens. An unknown key draws a dashed circle and warns, so a typo shows up in review instead of breaking the page, as does a row with a fifth cell or a keyword it doesn't take (skipped). A block with nothing to draw shows its source. A `← note` on any line is dropped (partials/fence/line.html). Obsidian shows the block as plain text. */}} {{- $slots := slice }} {{- $kind := "" }} {{- $caption := "" }} {{- $bar := false }} {{- range $n, $raw := split .Inner "\n" }} {{- $line := partialCached "fence/line.html" $raw $raw }} {{- $at := partial "fence/where.html" (dict "pos" $.Position "line" $n) }} {{- if not $line }}{{ continue }}{{ end }} {{- if hasPrefix $line "caption:" }} {{- $caption = trim (strings.TrimPrefix "caption:" $line) " " }} {{- continue }} {{- end }} {{- if hasPrefix $line "bar:" }} {{- $bar = in (slice "yes" "true") (trim (strings.TrimPrefix "bar:" $line) " ") }} {{- continue }} {{- end }} {{- if findRE `^[a-z][a-z-]*: ` $line 1 }} {{- warnf "formula [unknown-keyword]: skipped %q (takes caption: and bar:) at %q" $line $at }} {{- continue }} {{- end }} {{- $op := "" }} {{- with findRESubmatch `^(\+|:|→|->|=)\s+(.*)$` $line 1 }} {{- $op = index (index . 0) 1 }} {{- $line = index (index . 0) 2 }} {{- end }} {{- if eq $op "->" }}{{ $op = "→" }}{{ end }} {{- if and $op (ne $op "=") }} {{- if and $kind (ne $kind $op) }} {{- warnf "formula [mixed-markers]: %s after %s, one operator per block, at %q" $op $kind $at }} {{- else }} {{- $kind = $op }} {{- end }} {{- end }} {{- $f := slice }} {{- range split $line "|" }}{{ $f = $f | append (trim . " ") }}{{ end }} {{- if gt (len $f) 4 }} {{- warnf "formula [too-many-cells]: %d cells in %q (want icon | label | quantity | swaps) at %q" (len $f) $line $at }} {{- end }} {{- $slot := dict "op" $op "icons" (findRE `[a-z-]+` (index $f 0)) "label" "" "qty" "" "swaps" slice "at" $at }} {{- if ge (len $f) 2 }}{{ $slot = merge $slot (dict "label" (index $f 1)) }}{{ end }} {{- if ge (len $f) 3 }}{{ $slot = merge $slot (dict "qty" (index $f 2)) }}{{ end }} {{- if ge (len $f) 4 }} {{- $sw := slice }} {{- range split (index $f 3) "," }}{{ with trim . " " }}{{ $sw = $sw | append . }}{{ end }}{{ end }} {{- $slot = merge $slot (dict "swaps" $sw) }} {{- end }} {{- $slots = $slots | append $slot }} {{- end }} {{- if gt (len $slots) 5 }} {{- warnf "formula [over-limit]: %d slots; more than five won't fit the sidebar, at %q" (len $slots) (index $slots 5).at }} {{- end }} {{- if not $slots }} {{- warnf "formula [no-valid-lines]: nothing to draw, so the source shows, at %q" (partial "fence/where.html" (dict "pos" .Position "line" -1)) }} 
{{ .Inner }}
{{- else }} {{- $variant := "add" }} {{- if eq $kind ":" }}{{ $variant = "ratio" }}{{ else if eq $kind "→" }}{{ $variant = "seq" }}{{ end }} {{- $words := dict "+" "plus" ":" "to" "→" "then" "=" "makes" }} {{- $aria := slice }} {{- range $slots }} {{- $part := trim (printf "%s %s" .label .qty) " " }} {{- with .op }}{{ $part = printf "%s %s" (index $words .) $part }}{{ end }} {{- $aria = $aria | append $part }} {{- end }} {{- end }}
